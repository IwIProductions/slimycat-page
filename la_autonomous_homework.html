class Vehiculo:
    # Atributo de clase
    categoria = "Objeto Mecánico"

    def __init__(self, marca, modelo, color):
        self.marca = marca
        self.modelo = modelo
        self.color = color

    def mostrar_datos(self):
        print("\n----- VEHÍCULO -----")
        print(f"Marca : {self.marca}")
        print(f"Modelo: {self.modelo}")
        print(f"Color : {self.color}")

    def hacer_sonido(self):
        print(f"El {self.marca} {self.modelo} hace: ¡Brrmmm Brrmmm!")

    @classmethod
    def mostrar_categoria(cls):
        print("\nCategoría:", cls.categoria)


class Animal:
    # Atributo de clase
    reino = "Animal"

    def __init__(self, nombre, especie, edad):
        self.nombre = nombre
        self.especie = especie
        self.edad = edad

    def mostrar_datos(self):
        print("\n----- ANIMAL -----")
        print(f"Nombre : {self.nombre}")
        print(f"Especie: {self.especie}")
        print(f"Edad   : {self.edad} años")

    def hacer_sonido(self):
        sonidos = {
            "Perro": "Guau Guau",
            "Gato": "Miau",
            "Vaca": "Muuu",
            "Pato": "Cuac",
            "Caballo": "Relincho"
        }

        sonido = sonidos.get(self.especie, "Sonido desconocido")
        print(f"{self.nombre} hace: {sonido}")

    @classmethod
    def mostrar_reino(cls):
        print("\nReino:", cls.reino)
